    function createStars() {
      const container = document.getElementById('stars-container');
      if (!container) return;
      container.innerHTML = '';

      for (let i = 0; i < 220; i++) {
        const star = document.createElement('div');
        star.className = 'star';
        
        const size = Math.random() * 3.2 + 0.8;
        star.style.width = `${size}px`;
        star.style.height = `${size}px`;
        
        star.style.left = `${Math.random() * 100}%`;
        star.style.top = `${Math.random() * 85}%`;
        
        const duration = Math.random() * 2800 + 2200;
        star.style.animationDuration = `${duration}ms`;
        star.style.animationDelay = `-${Math.random() * 6000}ms`;
        
        if (Math.random() > 0.92) {
          star.style.boxShadow = '0 0 12px 3px rgba(255,255,255,0.95)';
        }
        
        container.appendChild(star);
      }
    }

    function initializeThemeToggle() {
      const toggle = document.getElementById('theme-toggle');
      const icon = document.getElementById('theme-icon');

      const savedTheme = localStorage.getItem('theme');
      const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
      const shouldBeDark = savedTheme === 'dark' || (savedTheme === null && prefersDark);

      function updateUI(isDark) {
        if (isDark) {
          document.documentElement.classList.add('dark');
          icon.classList.remove('fa-sun');
          icon.classList.add('fa-moon');
        } else {
          document.documentElement.classList.remove('dark');
          icon.classList.remove('fa-moon');
          icon.classList.add('fa-sun');
        }
      }

      // Initial state
      updateUI(shouldBeDark);

      // Toggle handler
      toggle.addEventListener('click', () => {
        const isNowDark = document.documentElement.classList.toggle('dark');
        updateUI(isNowDark);
        localStorage.setItem('theme', isNowDark ? 'dark' : 'light');
        if (isNowDark) createStars();
      });
    }

    function initializeNavbar() {
      const navbar = document.getElementById('navbar');
      window.addEventListener('scroll', () => {
        if (window.scrollY > 30) {
          navbar.classList.add('nav-scrolled', 'shadow-sm');
        } else {
          navbar.classList.remove('nav-scrolled', 'shadow-sm');
        }
      });
    }

    function initializeMobileMenu() {
      const menuBtn = document.getElementById('mobile-menu-btn');
      const mobileMenu = document.getElementById('mobile-menu');
      const mobileLinks = document.querySelectorAll('.mobile-nav-link');

      menuBtn.addEventListener('click', () => {
        const isHidden = mobileMenu.classList.contains('hidden');
        if (isHidden) {
          mobileMenu.classList.remove('hidden');
          menuBtn.innerHTML = '<i class="fa-solid fa-times text-lg"></i>';
        } else {
          mobileMenu.classList.add('hidden');
          menuBtn.innerHTML = '<i class="fa-solid fa-bars text-lg"></i>';
        }
      });

      mobileLinks.forEach(link => {
        link.addEventListener('click', () => {
          mobileMenu.classList.add('hidden');
          menuBtn.innerHTML = '<i class="fa-solid fa-bars text-lg"></i>';
        });
      });
    }

    function initializeSmoothScroll() {
      document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
          const targetId = this.getAttribute('href').substring(1);
          const target = document.getElementById(targetId);
          if (target) {
            e.preventDefault();
            const offset = 80;
            const elementPosition = target.getBoundingClientRect().top;
            const offsetPosition = elementPosition + window.scrollY - offset;
            window.scrollTo({ top: offsetPosition, behavior: 'smooth' });
          }
        });
      });
    }

    function initializeContactForm() {
      emailjs.init("xWHJIbmTLAix0i87H");

      const form = document.getElementById('contact-form');
      const successMessage = document.getElementById('success-message');
      const submitBtn = form.querySelector('button[type="submit"]');
      const submitText = document.getElementById('submit-text');
      const submitIcon = document.getElementById('submit-icon');

      form.addEventListener('submit', function (e) {
        e.preventDefault();

        const honeypot = form.querySelector('#website');
        if (honeypot && honeypot.value.trim() !== '') {
          form.classList.add('hidden');
          successMessage.classList.remove('hidden');
          setTimeout(() => {
            form.reset();
            form.classList.remove('hidden');
            successMessage.classList.add('hidden');
          }, 4200);
          return;
        }

        submitBtn.disabled = true;
        submitText.textContent = 'Sending...';
        submitIcon.classList.replace('fa-paper-plane', 'fa-spinner');
        submitIcon.classList.add('fa-spin');

        emailjs.sendForm('service_p2fz4ym', 'template_k5ry3sb', form)
          .then(() => {
            form.classList.add('hidden');
            successMessage.classList.remove('hidden');
            setTimeout(() => {
              form.reset();
              form.classList.remove('hidden');
              successMessage.classList.add('hidden');
              submitBtn.disabled = false;
              submitText.textContent = 'Send Message';
              submitIcon.classList.remove('fa-spin');
              submitIcon.classList.replace('fa-spinner', 'fa-paper-plane');
            }, 4200);
          })
          .catch((error) => {
            alert("Oops! Something went wrong. Please try again.");
            console.error("EmailJS Error:", error);
            submitBtn.disabled = false;
            submitText.textContent = 'Send Message';
            submitIcon.classList.remove('fa-spin');
            submitIcon.classList.replace('fa-spinner', 'fa-paper-plane');
          });
      });
    }

    function setCurrentYear() {
      document.getElementById('year').textContent = new Date().getFullYear();
    }

    function initializePortfolio() {
      initializeThemeToggle();
      initializeNavbar();
      initializeMobileMenu();
      initializeSmoothScroll();
      initializeContactForm();
      setCurrentYear();
      
      if (document.documentElement.classList.contains('dark')) {
        createStars();
      }
    }

    document.addEventListener('DOMContentLoaded', initializePortfolio);
